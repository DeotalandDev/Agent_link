cmake_minimum_required(VERSION 3.16)
if(CMAKE_SCRIPT_MODE_FILE)
    return()
endif()

# Per-board sdkconfig. config.json's "sdkconfig_append" is the single source of truth
# parse boards/<dir>/config.json and generate a defaults file layered on top of the shared sdkconfig.defaults
if(NOT DEFINED BOARD AND DEFINED ENV{BOARD})
    set(BOARD "$ENV{BOARD}")
endif()
set(SDKCONFIG_DEFAULTS "sdkconfig.defaults")
if(BOARD)
    set(_board_json "${CMAKE_CURRENT_LIST_DIR}/boards/${BOARD}/config.json")
    if(NOT EXISTS "${_board_json}")
        message(FATAL_ERROR "BOARD='${BOARD}' but ${_board_json} not found")
    elseif(CMAKE_VERSION VERSION_LESS 3.19)
        message(WARNING "CMake ${CMAKE_VERSION} < 3.19 cannot parse config.json; "
                        "set ${BOARD}'s sdkconfig via menuconfig")
    else()
        file(READ "${_board_json}" _json)
        string(JSON _append ERROR_VARIABLE _err GET "${_json}" builds 0 sdkconfig_append)
        if(_err)
            message(WARNING "no builds[0].sdkconfig_append in ${_board_json}: ${_err}")
        else()
            string(JSON _n LENGTH "${_append}")
            set(_lines "# generated from boards/${BOARD}/config.json - do not edit\n")
            if(_n GREATER 0)
                math(EXPR _last "${_n} - 1")
                foreach(_i RANGE 0 ${_last})
                    string(JSON _l GET "${_append}" ${_i})
                    string(APPEND _lines "${_l}\n")
                endforeach()
            endif()
            set(_gen "${CMAKE_BINARY_DIR}/board-${BOARD}.sdkconfig.defaults")
            file(WRITE "${_gen}" "${_lines}")
            list(APPEND SDKCONFIG_DEFAULTS "${_gen}")
            message(STATUS "agent_link: board '${BOARD}' - ${_n} sdkconfig option(s) from config.json")
        endif()
    endif()
endif()

include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(agent_link)
